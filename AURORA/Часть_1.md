**1. Базовая агрегация: Конверсия за последний месяц**

WITH leads_last_month AS (
    -- Выбираем лиды, созданные в прошлом месяце (относительно текущей даты)
    SELECT id, source_channel 
    FROM crm_leads
    WHERE date_create >= DATE_TRUNC('month', CURRENT_DATE) - INTERVAL '1 month'
      AND date_create < DATE_TRUNC('month', CURRENT_DATE)
),
deals_last_month AS (
    -- Считаем количество сделок по этим лидам
    SELECT l.source_channel,
           COUNT(l.id) AS total_leads,
           COUNT(d.id) AS deals_count
    FROM leads_last_month l
    LEFT JOIN crm_deals d ON l.id = d.lead_id
    GROUP BY l.source_channel
)
SELECT 
    source_channel,
    total_leads,
    deals_count,
    ROUND(
        CASE WHEN total_leads > 0 THEN deals_count::DECIMAL / total_leads * 100 ELSE NULL END, 2) AS conversion_rate_percent
FROM deals_last_month;


*2. CAC и ROMI за квартал**

-- Определяем начало текущего квартала
WITH quarter_start AS (
    SELECT DATE_TRUNC('quarter', CURRENT_DATE)::DATE AS q_start
),
spend_qtr AS (
    -- Суммируем расходы по каналам за квартал
    SELECT channel, SUM(spend) AS spend_total
    FROM marketing_spend
    WHERE date >= (SELECT q_start FROM quarter_start)
      AND date < CURRENT_DATE
    GROUP BY channel
),
revenue_qtr AS (
-- Суммируем выручку и считаем успешные сделки по каналам за квартал
    SELECT l.source_channel AS channel,
           COUNT(DISTINCT d.id) FILTER(WHERE d.stage_name = 'Закрыто успешно') AS successful_deals,
           SUM(d.revenue) FILTER(WHERE d.stage_name = 'Закрыто успешно') AS revenue_total
    FROM crm_deals d
    JOIN crm_leads l ON d.lead_id = l.id
    WHERE d.date_close >= (SELECT q_start FROM quarter_start)
      AND d.date_close < CURRENT_DATE
    GROUP BY l.source_channel
)
SELECT 
    COALESCE(r.channel, s.channel) AS channel,
    s.spend_total,
    r.successful_deals,
    r.revenue_total,
    
-- CAC: Стоимость привлечения клиента
    CASE WHEN r.successful_deals > 0 THEN s.spend_total / r.successful_deals ELSE NULL END AS cac,
    
-- ROMI: Возврат инвестиций (%)
    CASE WHEN s.spend_total > 0 THEN ((r.revenue_total - s.spend_total) / s.spend_total) * 100 ELSE NULL END AS romi_percent
FROM spend_qtr s
FULL OUTER JOIN revenue_qtr r USING(channel);

**3. Среднее время жизни лида по менеджерам**

WITH closed_deals AS (
-- Отфильтровываем только закрытые сделки
    SELECT 
        l.manager_id,
        l.date_create,
        d.date_close,
        
-- Разница во времени в секундах
        EXTRACT(EPOCH FROM (d.date_close - l.date_create)) AS seconds_to_close
    FROM crm_deals d
    INNER JOIN crm_leads l ON d.lead_id = l.id
    WHERE d.date_close IS NOT NULL
)
SELECT 
    manager_id,
    COUNT(*) AS deals_count,
    
-- Переводим среднее из секунд в дни
    ROUND(AVG(seconds_to_close) / (60 * 60 * 24), 2) AS avg_days_to_close
FROM closed_deals
GROUP BY manager_id
ORDER BY avg_days_to_close DESC;
